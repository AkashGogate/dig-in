// Device pills
function renderDevicePills() {

// Helpers
function fmtTime(s){ const m=Math.floor(s/60), sc=s%60; return `${m}:${String(sc).padStart(2,'0')}`; }
function timerClass(s){ if (s>=900) return 'crit'; if (s>=600) return 'warn'; return ''; }
function srcIcon(d){
  if (d.source==='camera') return `<span class="src" title="Seen by camera">${SVG.camera}</span>`;
  if (d.source==='tapped') return `<span class="src" title="Marked by hand">${SVG.hand}</span>`;
  return '';
}
function statusWord(st){ return st==='ready'?'Ready':st==='cooking'?'Cooking':'Queued'; }

// Dish row
function dishRow(d, camOffline) {
  const meta = [`Seat ${d.seat}`, d.zone, d.station, d.modifiers].filter(Boolean).join(' · ');
  return `
    <div class="dish">
      <div class="dish-main">
        <div class="dish-name">${d.name}${d.allergy?` <span class="allergy-pill">${d.allergy}</span>`:''}</div>
        <div class="dish-meta">${meta}${d.status==='cooking'&&d.est_min?` · ~${d.est_min}m`:''}</div>
      </div>
      <div class="dish-right">
        <span class="d-status ${d.status}"><span class="dot"></span>${statusWord(d.status)}${d.status==='ready'?srcIcon(d):''}</span>
        ${camOffline && d.status==='cooking' ? `<button class="btn-mark-ready" data-ready="${d.id}">Mark ready</button>` : ''}
      </div>
    </div>`;
}

// Ticket
function renderTicket(order) {
  const dishes = activeStation==='All' ? order.dishes : order.dishes.filter(d => d.station===activeStation);
  if (!dishes.length) return '';
  const readyN = dishes.filter(d => d.status==='ready').length;
  const allReady = readyN === dishes.length;
  const camOffline = api.getCameraStatus()==='offline';
  const tc = timerClass(order.elapsed_s || 0);
  const mode = order.delivery_mode==='all_together' ? 'All together' : 'As each is ready';
  const readyPill = allReady
    ? `<span class="ticket-ready-pill done">Ready</span>`
    : `<span class="ticket-ready-pill">${readyN}/${dishes.length} ready</span>`;

  return `
    <div class="ticket ${allReady?'allready':''}">
      <div class="ticket-head">
        <div class="ticket-head-top">
          <span class="ticket-table">Table ${order.table}</span>
          <span class="ticket-timer ${tc}">${fmtTime(order.elapsed_s||0)}</span>
        </div>
        <div class="ticket-head-sub">
          <span class="ticket-mode">${mode} · ${dishes.length} dish${dishes.length!==1?'es':''}</span>
          ${readyPill}
        </div>
      </div>
      <div class="ticket-dishes">${dishes.map(d => dishRow(d, camOffline)).join('')}</div>
    </div>`;
}

// Clarification
function renderClarification(c, orders) {
  const table = parseInt(c.zone.split('·')[0]);
  const order = orders.find(o => o.table===table);
  const opts = order ? order.dishes : [];
  return `
    <div class="clarify">
      <div class="clarify-head">${SVG.help} Camera · needs confirm</div>
      <div class="clarify-q">${c.text}</div>
      <div class="clarify-opts">
        ${opts.map(d => `<button class="btn-clarify" data-cd="${d.id}" data-ci="${c.id}">${d.name} <span class="z">Zone ${d.zone}</span></button>`).join('')}
        <button class="btn-clarify dismiss" data-cdismiss="${c.id}">Not one of ours</button>
      </div>
    </div>`;
}

// Zones
function renderZones(orders) {
  const el = document.getElementById('zoneView');
  if (!orders.length) { el.innerHTML = '<div class="empty-note">No active orders</div>'; return; }
  el.innerHTML = orders.map(o => {
    const dishes = activeStation==='All' ? o.dishes : o.dishes.filter(d => d.station===activeStation);
    if (!dishes.length) return '';
    const readyN = dishes.filter(d => d.status==='ready').length;
    const tiles = dishes.map(d => {
      const cls = d.status==='ready'?'ready':d.status==='cooking'?'cooking':'';
      const st = d.status==='ready'?'Ready':d.status==='cooking'?(d.est_min?`~${d.est_min}m`:'Cooking'):'Queued';
      return `<div class="zone-tile ${cls}"><div class="zt-num">${d.zone}</div><div class="zt-st">${st}</div></div>`;
    }).join('');
    return `<div class="zone-panel">
      <div class="zone-panel-head">
        <div class="zone-panel-title">Table ${o.table}</div>
        <div class="zone-panel-sub">${readyN}/${dishes.length} ready</div>
      </div>
      <div class="zone-tiles">${tiles}</div>
    </div>`;
  }).filter(Boolean).join('');
}

// Ingredients
async function renderIngredients() {
  const el = document.getElementById('ingredientsView');
  const ings = await api.getIngredients();
  el.innerHTML = `<div class="ing-note">Camera-tracked — expected for active tickets vs plates seen leaving the pass vs in stock.</div>` +
    ings.map(i => {
      const seenCls = i.seen < i.expected ? 'warn' : 'good';
      return `<div class="ing-row">
        <div class="ing-name">${i.name}</div>
        <div class="ing-stat"><div class="ing-num info">${i.expected}</div><div class="ing-lbl">Expected</div></div>
        <div class="ing-stat"><div class="ing-num ${seenCls}">${i.seen}</div><div class="ing-lbl">Seen leaving</div></div>
        <div class="ing-stat"><div class="ing-num info">${i.stock}</div><div class="ing-lbl">In stock</div></div>
      </div>`;
    }).join('');
}

// Bottom bar
function renderAlldayBar(orders) {
  const all = orders.flatMap(o => activeStation==='All' ? o.dishes : o.dishes.filter(d => d.station===activeStation));
  const r = all.filter(d=>d.status==='ready').length, c = all.filter(d=>d.status==='cooking').length, q = all.filter(d=>d.status==='queued').length;
  document.getElementById('alldayBar').innerHTML = `
    <div class="allday-stat"><span class="allday-num">${all.length}</span><span class="allday-lbl">All day</span></div>
    <div class="allday-stat"><span class="allday-num" style="color:var(--green)">${r}</span><span class="allday-lbl">Ready</span></div>
    <div class="allday-stat"><span class="allday-num" style="color:var(--orange)">${c}</span><span class="allday-lbl">Cooking</span></div>
    <div class="allday-stat"><span class="allday-num" style="color:var(--label3)">${q}</span><span class="allday-lbl">Queued</span></div>`;
}

// Main render
async function refresh() {
  const [orders, heard] = await Promise.all([api.getKitchenQueue(), api.getHeardOrders()]);
  const sorted = [...orders].sort((a,b) => (b.elapsed_s||0)-(a.elapsed_s||0));
  renderDevicePills();
  renderLastHeard();

  if (activeView==='tickets') {
    const tv = document.getElementById('ticketView');
    const clar = api.getClarifications().map(c => renderClarification(c, orders)).join('');
    const call = heard.map(renderCallout).join('');
    const tix  = sorted.map(renderTicket).join('');
    tv.innerHTML = clar + call + tix;
    tv.querySelectorAll('[data-confirm]').forEach(b => b.addEventListener('click', async () => { b.disabled=true; b.textContent='Confirmed'; await api.confirmHeardOrder(b.dataset.confirm); refresh(); }));
    tv.querySelectorAll('[data-ready]').forEach(b => b.addEventListener('click', () => api.markDishReady(b.dataset.ready)));
    tv.querySelectorAll('[data-cd]').forEach(b => b.addEventListener('click', () => { api.markDishReady(b.dataset.cd); api.resolveClarification(b.dataset.ci); }));
    tv.querySelectorAll('[data-cdismiss]').forEach(b => b.addEventListener('click', () => api.resolveClarification(b.dataset.cdismiss)));
  } else if (activeView==='zones') {
    renderZones(sorted);
  } else {
    renderIngredients();
  }
  renderAlldayBar(orders);
}

  .forEach(ev => on(ev, () => refresh()));

refresh();
startMockRealtime();
setInterval(() => { for (const o of api._mock.orders) o.elapsed_s = (o.elapsed_s||0)+1; refresh(); }, 1000);
