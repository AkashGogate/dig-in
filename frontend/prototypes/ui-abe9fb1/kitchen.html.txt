// ── Helpers ─────────────────────────────────────────────────────────────────
function fmtTime(s) { const m = Math.floor(s/60), sc = s%60; return `${m}:${String(sc).padStart(2,'0')}`; }
function timerClass(s) { if (s >= 600) return 'crit'; if (s >= 360) return 'warn'; return ''; }
function groupBySeats(dishes) { const seats = {}; for (const d of dishes) { (seats[d.seat] ||= []).push(d); } return seats; }

function srcBadge(d) {
  if (d.status !== 'ready' || !d.source) return '';
  if (d.source === 'camera') return `<span class="src-badge camera" title="Seen by camera">${SVG.camera}</span>`;
  if (d.source === 'tapped') return `<span class="src-badge tapped" title="Marked by hand">${SVG.hand}</span>`;
  return '';
}

// ── Render ticket ────────────────────────────────────────────────────────────
function renderTicket(order) {
  const dishes = activeStation === 'All' ? order.dishes : order.dishes.filter(d => d.station === activeStation);
  if (!dishes.length) return '';

  const tc = timerClass(order.elapsed_s || 0);
  const deliveryLabel = order.delivery_mode === 'all_together' ? 'All together' : 'As each is ready';
  const camOffline = api.getCameraStatus() === 'offline';
  const seats = groupBySeats(dishes);

  const seatHtml = Object.entries(seats).map(([seat, ds]) => `
    <div class="ticket-seat-group">
      <div class="ticket-seat-label">Seat ${seat}</div>
      ${ds.map(d => `
        <div class="ticket-dish-row">
          <div class="ticket-dish-main">
            <div class="ticket-dish-name">${d.name}${d.allergy ? ` <span class="allergy-pill" style="font-size:11px;padding:2px 7px">${d.allergy}</span>` : ''}</div>
            ${d.modifiers ? `<div class="ticket-dish-mod">${d.modifiers}</div>` : ''}
            <div class="ticket-dish-zone">
              ${srcBadge(d)}
              <span>Zone ${d.zone} · ${d.station}${d.status === 'ready' ? ' · <span style="color:var(--green)">READY</span>' : d.est_min ? ` · ~${d.est_min}m` : ''}</span>
            </div>
          </div>
          ${camOffline && d.status === 'cooking' ? `<button class="btn-mark-ready" data-ready="${d.id}">Mark ready</button>` : ''}
        </div>`).join('')}
    </div>`).join('');

  return `
    <div class="ticket-card confirmed" data-order="${order.id}">
      <div class="ticket-header">
        <div class="ticket-table-num">${order.table}</div>
        <div class="ticket-header-info">
          <div class="ticket-table-label">Table ${order.table}</div>
          <div class="ticket-meta">${dishes.length} dish${dishes.length !== 1 ? 'es' : ''} · ${dishes.filter(d=>d.status==='ready').length} ready</div>
        </div>
        <div class="ticket-timer ${tc}">${fmtTime(order.elapsed_s || 0)}</div>
      </div>
      <div class="delivery-chip">${deliveryLabel}</div>
      <div class="ticket-dishes">${seatHtml}</div>
    </div>`;
}

// ── Render zone view ─────────────────────────────────────────────────────────
function renderZones(orders) {
  const container = document.getElementById('zoneView');
  if (!orders.length) { container.innerHTML = '<div style="color:var(--label3);padding:40px;text-align:center;font:400 17px/22px var(--font)">No active orders</div>'; return; }
  container.innerHTML = orders.map(order => {
    const dishes = activeStation === 'All' ? order.dishes : order.dishes.filter(d => d.station === activeStation);
    if (!dishes.length) return '';
    const tiles = dishes.map(d => {
      const cls = d.status === 'ready' ? 'ready' : d.status === 'cooking' ? 'cooking' : 'queued';
      const statusTxt = d.status === 'ready' ? 'READY' : d.status === 'cooking' ? (d.est_min ? `~${d.est_min}m` : 'COOKING') : 'QUEUED';
      return `<div class="zone-tile ${cls}">${srcBadge(d)}<div class="zone-tile-num">${d.zone}</div><div class="zone-tile-status">${statusTxt}</div></div>`;
    }).join('');
    return `<div class="zone-row"><div class="zone-row-label">Table ${order.table}</div><div class="zone-tiles">${tiles}</div></div>`;
  }).filter(Boolean).join('');
}

    <div class="allday-sep"></div>
    <div class="allday-stat"><div class="allday-num" style="color:var(--label3)">${queued}</div><div class="allday-lbl">Queued</div></div>`;
  document.getElementById('allDaySummary').innerHTML = `All day: <span>${total}</span> dishes`;
}

// ── Main render ──────────────────────────────────────────────────────────────
async function refresh() {
  const [orders, heard] = await Promise.all([api.getKitchenQueue(), api.getHeardOrders()]);
  const sorted = [...orders].sort((a, b) => (b.elapsed_s || 0) - (a.elapsed_s || 0));

  renderDevicePills();
  renderLastHeard();

  if (activeView === 'tickets') {
    const tv = document.getElementById('ticketView');
    const clarifyHtml = api.getClarifications().map(c => renderClarification(c, orders)).join('');
    const heardHtml = heard.map(renderHeard).join('');
    const ticketHtml = sorted.map(renderTicket).join('');
    tv.innerHTML = clarifyHtml + heardHtml + ticketHtml;

    tv.querySelectorAll('[data-confirm]').forEach(btn => {
      btn.addEventListener('click', async () => { btn.disabled = true; btn.textContent = 'Confirmed'; await api.confirmHeardOrder(btn.dataset.confirm); refresh(); });
    });
    tv.querySelectorAll('[data-ready]').forEach(btn => {
      btn.addEventListener('click', () => api.markDishReady(btn.dataset.ready));
    });
    tv.querySelectorAll('[data-clarify-dish]').forEach(btn => {
      btn.addEventListener('click', () => { api.markDishReady(btn.dataset.clarifyDish); api.resolveClarification(btn.dataset.clarifyId); });
    });
    tv.querySelectorAll('[data-clarify-dismiss]').forEach(btn => {
      btn.addEventListener('click', () => api.resolveClarification(btn.dataset.clarifyDismiss));
    });
  } else if (activeView === 'zones') {
    renderZones(sorted);
  } else {
    renderIngredients();
  }

  renderAlldayBar(orders);
}

// ── Realtime events ──────────────────────────────────────────────────────────
['dish_ready', 'dish_update', 'heard_order', 'heard_confirmed', 'last_heard',
 'last_heard_undo', 'mic_status', 'camera_status', 'clarification', 'clarification_resolved'
].forEach(ev => on(ev, () => refresh()));

refresh();
